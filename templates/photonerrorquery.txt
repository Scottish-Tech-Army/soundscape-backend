AzureDiagnostics
| where Category == "FrontDoorAccessLog"
| where requestUri_s startswith "photon." or requestUri_s startswith "photontest."
| extend status = toint(httpStatusCode_s)
| extend isValid = tostring(requestUri_s) contains "/photon/"
| summarize
    ValidRequests   = countif(isValid and status != 200),
    InvalidRequests = countif(not(isValid) and status != 200)
    by TimeGenerated = bin(TimeGenerated, 1h);
